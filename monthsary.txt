<script>
(function(){
  // basic router (tab buttons)
  const tabs = document.querySelectorAll('.tab');
  const sections = {home:document.getElementById('home'), timeline:document.getElementById('timeline'), slideshow:document.getElementById('slideshow')};
  function go(route){
    tabs.forEach(t=>t.classList.toggle('active', t.dataset.route===route));
    Object.keys(sections).forEach(k=>sections[k].style.display = (k===route ? 'block' : 'none'));
  }
  tabs.forEach(t=>t.addEventListener('click', ()=>go(t.dataset.route)));

  // start date (months counter)
  const startInput = document.getElementById('startDate');
  const monthsEl = document.getElementById('months');
  const sinceEl = document.getElementById('since');
  const openHeart = document.getElementById('openHeart');
  const editStart = document.getElementById('editStart');

  function loadStart(){
    const s = localStorage.getItem('monthsary_start') || '';
    if(s) startInput.value = s;
    updateCounter();
  }
  function saveStart(){
    const v = startInput.value;
    if(v) localStorage.setItem('monthsary_start', v);
    updateCounter();
  }
  function monthsBetween(a,b){
    // full months elapsed from date a (YYYY-MM-DD) to b (Date)
    const start = new Date(a);
    if(isNaN(start)) return 0;
    const y = b.getFullYear() - start.getFullYear();
    let m = b.getMonth() - start.getMonth();
    let months = y*12 + m;
    if(b.getDate() < start.getDate()) months -= 1;
    return Math.max(0, months);
  }
  function daysSinceLastMonth(a,b){
    const start = new Date(a);
    const months = monthsBetween(a,b);
    const monthDate = new Date(start.getFullYear(), start.getMonth()+months, start.getDate());
    const diff = Math.floor((b - monthDate)/(1000*60*60*24));
    return diff;
  }
  function updateCounter(){
    const s = localStorage.getItem('monthsary_start') || startInput.value;
    const now = new Date();
    if(!s){ monthsEl.textContent = '—'; sinceEl.textContent = 'No start date set'; return; }
    const months = monthsBetween(s, now);
    monthsEl.textContent = months;
    const days = daysSinceLastMonth(s, now);
    sinceEl.textContent = `${months} months · ${days} days since last monthsary`;
  }
  document.getElementById('saveStart').addEventListener('click', ()=>{ saveStart(); toast('Saved start date'); });
  editStart.addEventListener('click', ()=>{ startInput.focus() });
  openHeart.addEventListener('click', ()=>{
    // tiny animation: flash tabs to slideshow maybe
    const route = location.hash.replace('#','') || 'slideshow';
    go('slideshow');
  });
  loadStart();
  setInterval(updateCounter, 1000*30);

  // timeline storage
  const timelineList = document.getElementById('timelineList');
  function loadTimeline(){
    const raw = localStorage.getItem('monthsary_timeline');
    let arr = raw ? JSON.parse(raw) : [];
    renderTimeline(arr);
  }
  function renderTimeline(arr){
    timelineList.innerHTML = '';
    if(!arr.length){ timelineList.innerHTML = '<div style="text-align:center;color:var(--muted)">No moments yet</div>'; return; }
    arr.sort((a,b)=> new Date(b.date) - new Date(a.date));
    arr.forEach(it=>{
      const el = document.createElement('div'); el.className='item';
      el.innerHTML = `<div class="dot"></div><div><strong>${escapeHtml(it.title)}</strong><div style="color:var(--muted);font-size:13px">${formatDate(it.date)} · ${escapeHtml(it.note || '')}</div></div>`;
      timelineList.appendChild(el);
    });
  }
  document.getElementById('addMoment').addEventListener('click', ()=>{
    const t = document.getElementById('eventTitle').value.trim();
    const d = document.getElementById('eventDate').value;
    const n = document.getElementById('eventNote').value.trim();
    if(!t||!d) return toast('Please add a title and date');
    const raw = localStorage.getItem('monthsary_timeline');
    const arr = raw? JSON.parse(raw):[];
    arr.push({title:t,date:d,note:n});
    localStorage.setItem('monthsary_timeline', JSON.stringify(arr));
    document.getElementById('eventTitle').value=''; document.getElementById('eventDate').value=''; document.getElementById('eventNote').value='';
    toast('Moment added'); loadTimeline();
  });
  document.getElementById('clearMoments').addEventListener('click', ()=>{ if(confirm('Clear all moments?')){ localStorage.removeItem('monthsary_timeline'); loadTimeline(); } });
  loadTimeline();

  // slideshow
  const slidesEl = document.getElementById('slides');
  const thumbs = document.getElementById('thumbs');
  let images = JSON.parse(localStorage.getItem('monthsary_images')||'[]');
  let idx = 0;
  function renderSlides(){
    slidesEl.innerHTML = '';
    thumbs.innerHTML = '';
    if(images.length===0){ slidesEl.innerHTML = '<div class="slide"><div style="color:var(--muted)">No images yet — upload to create slideshow</div></div>'; return; }
    images.forEach((src,i)=>{
      const s = document.createElement('div'); s.className='slide';
      const img = document.createElement('img'); img.src = src; s.appendChild(img); slidesEl.appendChild(s);
      const th = document.createElement('div'); th.className='thumb'; if(i===idx) th.classList.add('active'); th.innerHTML = `<img src="${src}" style="width:100%;height:100%;object-fit:cover">`;
      th.addEventListener('click', ()=>{ idx = i; updateSlide(); });
      thumbs.appendChild(th);
    });
    updateSlide();
  }
  function updateSlide(){
    slidesEl.style.transform = `translateX(${-idx*100}%)`;
    Array.from(thumbs.children).forEach((t,i)=>t.classList.toggle('active', i===idx));
  }
  document.getElementById('next').addEventListener('click', ()=>{ if(images.length===0) return; idx = (idx+1)%images.length; updateSlide(); });
  document.getElementById('prev').addEventListener('click', ()=>{ if(images.length===0) return; idx = (idx-1+images.length)%images.length; updateSlide(); });
  document.getElementById('imageUpload').addEventListener('change', async (e)=>{
    const files = Array.from(e.target.files).slice(0,20);
    for(const f of files){
      const data = await fileToDataURL(f);
      images.push(data);
    }
    localStorage.setItem('monthsary_images', JSON.stringify(images));
    renderSlides();
  });
  document.getElementById('clearImages').addEventListener('click', ()=>{ if(confirm('Clear all slideshow images?')){ images=[]; localStorage.removeItem('monthsary_images'); renderSlides(); } });
  function fileToDataURL(f){ return new Promise((res,rej)=>{ const r=new FileReader(); r.onload=()=>res(r.result); r.onerror=rej; r.readAsDataURL(f); }); }
  renderSlides();

  // little helper functions
  function toast(txt){
    const el = document.createElement('div'); el.textContent=txt; el.style.position='fixed'; el.style.left='50%'; el.style.transform='translateX(-50%)'; el.style.bottom='22px'; el.style.padding='10px 14px'; el.style.borderRadius='12px'; el.style.background='rgba(60,30,40,0.9)'; el.style.color='white'; el.style.zIndex=9999; document.body.appendChild(el);
    setTimeout(()=>{ el.style.transition='all 220ms'; el.style.opacity=0; el.style.transform='translateX(-50%) translateY(8px)'; setTimeout(()=>el.remove(),240); },1200);
  }
  function escapeHtml(s){ return s.replace(/[&<>\"]/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c])); }
  function formatDate(d){ const dt=new Date(d); return dt.toLocaleDateString(undefined,{year:'numeric',month:'short',day:'numeric'}); }

  // small UX niceties: support hash routing
  function initFromHash(){ const h = location.hash.replace('#',''); if(h && sections[h]) go(h); }
  window.addEventListener('hashchange', initFromHash); initFromHash();

  // ensure counter loads after start date load
  if(!localStorage.getItem('monthsary_start')){
    // prefill with example date (you can change)
    const sample = new Date(); sample.setMonth(sample.getMonth()-1); // 1 month ago
    startInput.value = sample.toISOString().slice(0,10);
  }
  saveStart();
})();
</script>
